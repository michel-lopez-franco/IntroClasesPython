6.1 ¿Qué es el Polimorfismo?

Polimorfismo significa “muchas formas”. Es la capacidad de objetos de diferentes clases de responder al mismo método de manera diferente.

6.1.1 Analogía del mundo real

Piensa en el comando “habla”:

  • Un perro responde: “¡Guau!”
  • Un gato responde: “¡Miau!”
  • Un pato responde: “¡Cuac!”

Todos responden al mismo comando, pero cada uno a su manera. Eso es polimorfismo.

6.2 Polimorfismo Básico con Herencia

6.2.1 La Magia del Polimorfismo

No necesitamos saber el tipo exacto del objeto. Solo necesitamos saber que tiene el método hacer_sonido().

6.2.2 El despacho polimórfico, paso a paso

En cada vuelta del ciclo, la variable animal apunta a un objeto distinto. Python no mira el nombre de la variable sino el tipo real del objeto para decidir qué versión de hacer_sonido() ejecutar.

6.3 Duck Typing




“Si camina como un pato y grazna como un pato, entonces debe ser un pato.”




En Python, el polimorfismo no requiere herencia. Solo necesitas que los objetos tengan el mismo método.
















6.4 Polimorfismo con Diferentes Clases


Un ejemplo más práctico: diferentes formas de pago




class TarjetaCredito:
    def __init__(self, numero, titular):
        self.numero = numero
        self.titular = titular
    
    def procesar_pago(self, monto):
        print(f"Procesando ${monto} con tarjeta de crédito")
        print(f"Titular: {self.titular}")
        print(f"Últimos 4 dígitos: {self.numero[-4:]}")
        return True

class PayPal:
    def __init__(self, email):
        self.email = email
    
    def procesar_pago(self, monto):
        print(f"Procesando ${monto} con PayPal")
        print(f"Email: {self.email}")
        return True

class Efectivo:
    def procesar_pago(self, monto):
        print(f"Recibiendo ${monto} en efectivo")
        print("Por favor, entregue el dinero al cajero")
        return True

class Carrito:
    def __init__(self):
        self.total = 0
    
    def agregar_monto(self, monto):
        self.total += monto
    
    # Este método acepta CUALQUIER forma de pago
    def pagar(self, metodo_pago):
        print(f"\nTotal a pagar: ${self.total}")
        print("-" * 40)
        
        if metodo_pago.procesar_pago(self.total):
            print("✓ Pago exitoso")
            self.total = 0
        else:
            print("✗ Pago fallido")

class SMS:
    def __init__(self, numero):
        self.numero = numero
    
    def enviar(self, mensaje):
        # Limitar a 160 caracteres
        mensaje_corto = mensaje[:160]
        print(f"📱 Enviando SMS a {self.numero}")
        print(f"Mensaje: {mensaje_corto}")
        if len(mensaje) > 160:
            print("(Mensaje truncado)")
        print()

class Push:
    def __init__(self, dispositivo_id):
        self.dispositivo_id = dispositivo_id
    
    def enviar(self, mensaje):
        print(f"🔔 Enviando notificación push al dispositivo {self.dispositivo_id}")
        print(f"Mensaje: {mensaje}")
        print()

class SistemaNotificaciones:
    def __init__(self):
        self.canales = []
    
    def agregar_canal(self, canal):
        self.canales.append(canal)
    
    def notificar_todos(self, mensaje):
        print("=" * 50)
        print("Enviando notificaciones...")
        print("=" * 50)

            # Polimorfismo: cada canal envía a su manera
            canal.enviar(mensaje)

6.7 Ejemplo: Sistema de Archivos




class Archivo:
    def __init__(self, nombre, tamaño):
        self.nombre = nombre
        self.tamaño = tamaño  # en KB
    
    def abrir(self):
        print(f"Abriendo archivo: {self.nombre}")
    
    def obtener_tamaño(self):
        return self.tamaño
    
    def obtener_info(self):
        return f"{self.nombre} ({self.tamaño} KB)"

class ArchivoImagen(Archivo):
    def __init__(self, nombre, tamaño, resolucion):
        super().__init__(nombre, tamaño)
        self.resolucion = resolucion
    
    def abrir(self):
        print(f"🖼️  Abriendo imagen: {self.nombre}")
        print(f"Resolución: {self.resolucion}")
    
    def obtener_info(self):
        return f"{self.nombre} - {self.resolucion} ({self.tamaño} KB)"

class ArchivoAudio(Archivo):
    def __init__(self, nombre, tamaño, duracion):
        super().__init__(nombre, tamaño)
        self.duracion = duracion  # en segundos
    
    def abrir(self):
        print(f"🎵 Reproduciendo audio: {self.nombre}")
        print(f"Duración: {self.duracion} segundos")
    
    def obtener_info(self):
        return f"{self.nombre} - {self.duracion}s ({self.tamaño} KB)"

class ArchivoVideo(Archivo):
    def __init__(self, nombre, tamaño, duracion, resolucion):
        super().__init__(nombre, tamaño)
        self.duracion = duracion
        self.resolucion = resolucion
    
    def abrir(self):
        print(f"🎬 Reproduciendo video: {self.nombre}")
        print(f"Duración: {self.duracion}s - Resolución: {self.resolucion}")
    
    def obtener_info(self):
        return f"{self.nombre} - {self.resolucion} - {self.duracion}s ({self.tamaño} KB)"

class Carpeta:
    def __init__(self, nombre):
        self.nombre = nombre
        self.contenido = []
    
    def agregar(self, item):
        self.contenido.append(item)
    
    def listar(self):
        print(f"\n📁 Carpeta: {self.nombre}")
        print("-" * 50)
        for item in self.contenido:
            print(f"  • {item.obtener_info()}")
    
    def abrir_todo(self):
        print(f"\nAbriendo archivos de: {self.nombre}")
        print("=" * 50)
        for item in self.contenido:
            # Polimorfismo: cada archivo se abre a su manera
            item.abrir()
            print()
    
    def obtener_tamaño_total(self):
        total = sum(item.obtener_tamaño() for item in self.contenido)
        return total

# Crear archivos de diferentes tipos
foto1 = ArchivoImagen("vacaciones.jpg", 2500, "1920x1080")
foto2 = ArchivoImagen("perfil.png", 150, "500x500")
cancion = ArchivoAudio("favorita.mp3", 3500, 180)

documento = Archivo("readme.txt", 5)

# Crear carpeta y agregar archivos
mi_carpeta = Carpeta("Mis Documentos")
mi_carpeta.agregar(foto1)
mi_carpeta.agregar(foto2)
mi_carpeta.agregar(cancion)
mi_carpeta.agregar(video)
mi_carpeta.agregar(documento)

# Listar contenido
mi_carpeta.listar()
print(f"\nTamaño total: {mi_carpeta.obtener_tamaño_total()} KB")

# Abrir todos los archivos (polimorfismo en acción)
mi_carpeta.abrir_todo()

import math

class Circulo:
    def __init__(self, radio):
        self.radio = radio
    def calcular_area(self):
        return math.pi * self.radio ** 2
    def calcular_perimetro(self):
        return 2 * math.pi * self.radio

class Rectangulo:
    def __init__(self, ancho, alto):
        self.ancho, self.alto = ancho, alto
    def calcular_area(self):
        return self.ancho * self.alto
    def calcular_perimetro(self):
        return 2 * (self.ancho + self.alto)

class Triangulo:
    def __init__(self, a, b, c):
        self.a, self.b, self.c = a, b, c
    def calcular_perimetro(self):
        return self.a + self.b + self.c
    def calcular_area(self):  # Fórmula de Herón
        s = self.calcular_perimetro() / 2
        return math.sqrt(s * (s - self.a) * (s - self.b) * (s - self.c))

def mostrar_info_forma(forma):
    print(f"{type(forma).__name__:<11} área={forma.calcular_area():7.2f}  "
          f"perímetro={forma.calcular_perimetro():6.2f}")

formas = [Circulo(2), Rectangulo(3, 4), Triangulo(3, 4, 5)]
for f in formas:
    mostrar_info_forma(f)
print(f"Área total: {sum(f.calcular_area() for f in formas):.2f}")

class Coche:
    def viajar(self, distancia):
        print(f"🚗 Conduciendo {distancia} km por carretera")
    def obtener_velocidad_promedio(self):
        return 90

class Bicicleta:
    def viajar(self, distancia):
        print(f"🚲 Pedaleando {distancia} km")
    def obtener_velocidad_promedio(self):
        return 18

class Avion:
    def viajar(self, distancia):
        print(f"✈️ Volando {distancia} km")
    def obtener_velocidad_promedio(self):
        return 800

class Barco:
    def viajar(self, distancia):
        print(f"🚢 Navegando {distancia} km")
    def obtener_velocidad_promedio(self):
        return 35

def planear_viaje(transporte, distancia):
    transporte.viajar(distancia)
    horas = distancia / transporte.obtener_velocidad_promedio()
    print(f"   {type(transporte).__name__}: tiempo estimado {horas:.1f} h")

for t in [Coche(), Bicicleta(), Avion(), Barco()]:
    planear_viaje(t, 540)  # Guadalajara → CDMX, aprox.

Crea una jerarquía de empleados con diferentes formas de calcular salario:

Todos deben tener:


    

  • Método calcular_salario()
    • 

  • Método generar_recibo() que muestra nombre, puesto y salario
    • 



Crea una función procesar_nomina(empleados) que genere recibos para todos.






















TipSolución del ejercicio 3













class Empleado:
    puesto = "Empleado"
    def __init__(self, nombre, salario_fijo):
        self.nombre = nombre
        self.salario_fijo = salario_fijo
    def calcular_salario(self):
        return self.salario_fijo
    def generar_recibo(self):
        print(f"{self.nombre:<8} {self.puesto:<16} ${self.calcular_salario():>10,.2f}")

class EmpleadoPorHoras(Empleado):
    puesto = "Por horas"
    def __init__(self, nombre, horas, tarifa_por_hora):
        super().__init__(nombre, 0)
        self.horas = horas
        self.tarifa_por_hora = tarifa_por_hora
    def calcular_salario(self):
        return self.horas * self.tarifa_por_hora

class Vendedor(Empleado):
    puesto = "Vendedor"
    def __init__(self, nombre, salario_base, ventas, comision):
        super().__init__(nombre, salario_base)
        self.ventas = ventas
        self.comision = comision
    def calcular_salario(self):
        return self.salario_fijo + self.ventas * self.comision

class Gerente(Empleado):
    puesto = "Gerente"
    def __init__(self, nombre, salario_base, bono, ventas_equipo, porcentaje):
        super().__init__(nombre, salario_base)
        self.bono = bono
        self.ventas_equipo = ventas_equipo
        self.porcentaje = porcentaje
    def calcular_salario(self):
        return self.salario_fijo + self.bono + self.ventas_equipo * self.porcentaje

def procesar_nomina(empleados):
    total = 0
    for e in empleados:
        e.generar_recibo()   # Polimorfismo: cada uno calcula a su manera
        total += e.calcular_salario()
    print(f"{'TOTAL':<25} ${total:>10,.2f}")

    EmpleadoPorHoras("Iván", 80, 120),
    Vendedor("Marta", 8000, 150000, 0.03),
    Gerente("Jorge", 25000, 5000, 400000, 0.01),
])

import random

random.seed(7)  # Semilla fija para que el resultado sea reproducible

class Personaje:
    def __init__(self, nombre, vida=100, nivel=1):
        self.nombre = nombre
        self.vida = vida
        self.nivel = nivel

    # "Métodos abstractos": las subclases deben sobrescribirlos
    def atacar(self):
        raise NotImplementedError
    def habilidad_especial(self):
        raise NotImplementedError

    def defender(self, daño):
        self.vida = max(0, self.vida - daño)

    def esta_vivo(self):
        return self.vida > 0

    def info(self):
        print(f"{self.nombre} ({type(self).__name__}) vida={self.vida} nivel={self.nivel}")

class Guerrero(Personaje):
    def atacar(self):
        return random.randint(12, 18)
    def defender(self, daño):
        super().defender(int(daño * 0.7))  # Armadura: recibe 30% menos
    def habilidad_especial(self):
        print(f"  {self.nombre} usa ¡Golpe devastador!")
        return 30

class Mago(Personaje):
    def atacar(self):
        return random.randint(8, 14)
    def habilidad_especial(self):
        print(f"  {self.nombre} lanza ¡Bola de fuego!")
        return 40

class Arquero(Personaje):
    def atacar(self):
        return random.randint(10, 15)
    def defender(self, daño):
        if random.random() < 0.3:
            print(f"  {self.nombre} esquiva el ataque")
            return
        super().defender(daño)
    def habilidad_especial(self):
        print(f"  {self.nombre} dispara ¡Lluvia de flechas!")
        return 25

class Curandero(Personaje):
    def atacar(self):
        return random.randint(5, 9)
    def habilidad_especial(self):
        self.vida = min(100, self.vida + 25)
        print(f"  {self.nombre} se cura (+25) → vida {self.vida}")
        return 0

def simular_batalla(p1, p2):
    atacante, defensor = p1, p2
    turno = 1
    while p1.esta_vivo() and p2.esta_vivo():
        if random.random() < 0.2:
            daño = atacante.habilidad_especial()
        else:
            daño = atacante.atacar()
        if daño:
            defensor.defender(daño)
            print(f"T{turno}: {atacante.nombre} → {defensor.nombre} ({daño}); "
                  f"vida de {defensor.nombre}: {defensor.vida}")
        atacante, defensor = defensor, atacante
        turno += 1
    ganador = p1 if p1.esta_vivo() else p2
    print(f"🏆 Gana {ganador.nombre}")

simular_batalla(Guerrero("Thor"), Mago("Merlín"))
print()
simular_batalla(Arquero("Robin"), Curandero("Aria"))
